threaded_lru.py
A thread-safe LRU cache with per-entry TTL, size-based eviction,
hit/miss statistics, and a memoization decorator.
"""

import functools
import threading
import time
from collections import OrderedDict
from dataclasses import dataclass, field
from typing import Any, Callable, Dict, Generic, Hashable, Iterator, List, Optional, Tuple, TypeVar

__all__ = [
    "LRUCache",
    "CacheStats",
    "cached",
    "memoize",
]

K = TypeVar("K", bound=Hashable)
V = TypeVar("V")


@dataclass(frozen=True)
class CacheStats:
    """Immutable snapshot of cache statistics."""
    hits: int = 0
    misses: int = 0
    evictions: int = 0
    expired: int = 0

    @property
    def accesses(self) -> int:
        return self.hits + self.misses

    @property
    def hit_rate(self) -> float:
        accesses = self.accesses
        if accesses == 0:
            return 0.0
        return self.hits / accesses


class LRUCache(Generic[K, V]):
    """A thread-safe Least-Recently-Used cache.

    Features:
      * Bounded size (``max_size``) with LRU eviction.
      * Optional per-entry time-to-live (TTL) in seconds.
      * Thread-safe via a reentrant lock.
      * Hit/miss/eviction/expiry statistics.
    """

    def __init__(
        self,
        max_size: int = 512,
        ttl: Optional[float] = None,
        clock: Callable[[], float] = time.monotonic,
    ) -> None:
        if max_size <= 0:
            raise ValueError("max_size must be a positive integer")
        if ttl is not None and ttl <= 0:
            raise ValueError("ttl must be positive or None")

        self._max_size = max_size
        self._default_ttl = ttl
        self._clock = clock

        self._store: "OrderedDict[K, Tuple[V, Optional[float]]]" = OrderedDict()
        self._lock = threading.RLock()

        # Statistics
        self._hits = 0
        self._misses = 0
        self._evictions = 0
        self._expired = 0

    # -- internal helpers -------------------------------------------------

    def _is_expired(self, expiry: Optional[float], now: float) -> bool:
        return expiry is not None and now >= expiry

    def _evict_if_needed(self) -> None:
        """Evict LRU entries until within ``max_size``. Caller holds lock."""
        while len(self._store) > self._max_size:
            self._store.popitem(last=False)
            self._evictions += 1

    def _purge_expired(self, now: float) -> None:
        """Remove all expired entries. Caller holds lock."""
        expired_keys = [
            key for key, (_value, expiry) in self._store.items()
            if self._is_expired(expiry, now)
        ]
        for key in expired_keys:
            del self._store[key]
            self._expired += 1

    # -- public API -------------------------------------------------------

    def get(self, key: K, default: Optional[V] = None) -> Optional[V]:
        with self._lock:
            now = self._clock()
            if key in self._store:
                value, expiry = self._store[key]
                if self._is_expired(expiry, now):
                    del self._store[key]
                    self._expired += 1
                    self._misses += 1
                    return default
                # Mark as most recently used.
                self._store.move_to_end(key)
                self._hits += 1
                return value  # type: ignore[return-value]
            self._misses += 1
            return default

    def set(self, key: K, value: V, ttl: Optional[float] = None) -> None:
        with self._lock:
            now = self._clock()
            entry_ttl = ttl if ttl is not None else self._default_ttl
            expiry = None if entry_ttl is None else now + entry_ttl
            if key in self._store:
                self._store[key] = (value, expiry)
                self._store.move_to_end(key)
            else:
                self._store[key] = (value, expiry)
            self._evict_if_needed()

    def delete(self, key: K) -> bool:
        with self._lock:
            if key in self._store:
                del self._store[key]
                return True
            return False

    def pop(self, key: K, default: Optional[V] = None) -> Optional[V]:
        with self._lock:
            try:
                value, _expiry = self._store.pop(key)
                return value
            except KeyError:
                return default

    def clear(self) -> None:
        with self._lock:
            self._store.clear()

    def __getitem__(self, key: K) -> V:
        value = self.get(key)
        if value is None:
            raise KeyError(key)
        return value

    def __setitem__(self, key: K, value: V) -> None:
        self.set(key, value)

    def __contains__(self, key: K) -> bool:
        with self._lock:
            now = self._clock()
            if key not in self._store:
                return False
            _value, expiry = self._store[key]
            if self._is_expired(expiry, now):
                del self._store[key]
                self._expired += 1
                return False
            return True

    def __len__(self) -> int:
        with self._lock:
            return len(self._store)

    def __iter__(self) -> Iterator[K]:
        with self._lock:
            # Snapshot keys to avoid mutation during iteration.
            return list(self._store.keys()).__iter__()

    def items(self) -> List[Tuple[K, V]]:
        with self._lock:
            now = self._clock()
            result: List[Tuple[K, V]] = []
            for key, (value, expiry) in self._store.items():
                if self._is_expired(expiry, now):
                    del self._store[key]
                    self._expired += 1
                    continue
                result.append((key, value))
            return result

    @property
    def max_size(self) -> int:
        return self._max_size

    @property
    def ttl(self) -> Optional[float]:
        return self._default_ttl

    @property
    def stats(self) -> CacheStats:
        with self._lock:
            return CacheStats(
                hits=self._hits,
                misses=self._misses,
                evictions=self._evictions,
                expired=self._expired,
            )

    def __repr__(self) -> str:
        return (
            f"{type(self).__name__}(max_size={self._max_size}, "
            f"ttl={self._default_ttl}, size={len(self._store)})"
        )
