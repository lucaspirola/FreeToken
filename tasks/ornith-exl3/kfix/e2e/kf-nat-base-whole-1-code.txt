"""
thread_lru_cache.py
===================

A thread-safe Least-Recently-Used (LRU) cache with per-entry time-to-live
(TTL), size-based eviction, hit/miss statistics, and a decorator for
memoizing functions.

Overview
--------
This module provides two public building blocks:

1. :class:`LRUCache`
   A generic, thread-safe LRU cache.  It supports:

   * a maximum number of entries (``maxsize``); when the cache is full the
     least-recently-used entry is evicted to make room,
   * an optional per-entry time-to-live (``ttl``); entries older than the
     configured TTL are treated as expired and reported as cache misses,
   * hit/miss statistics and a computed hit rate,
   * a fully thread-safe implementation backed by a re-entrant lock.

2. :func:`cached`
   A decorator factory that memoizes a function using an :class:`LRUCache`.
   The wrapped function exposes ``cache_info()`` and ``cache_clear()`` helpers
   mirroring :func:`functools.lru_cache`.

Example
-------
>>> from thread_lru_cache import cached, LRUCache
>>> @cached(maxsize=128, ttl=60)
... def fibonacci(n):
...     if n < 2:
...         return n
...     return fibonacci(n - 1) + fibonacci(n - 2)
>>> fibonacci(10)
55
"""

    def __init__(self, hits: int, misses: int, size: int, maxsize: Optional[int], hit_rate: float) -> None:
        self.hits = hits
        self.misses = misses
        self.size = size
        self.maxsize = maxsize
        self.hit_rate = hit_rate

    @property
    def total(self) -> int:
        return self.hits + self.misses

    def as_dict(self) -> dict:
        return {
            "hits": self.hits,
            "misses": self.misses,
            "size": self.size,
            "maxsize": self.maxsize,
            "hit_rate": self.hit_rate,
        }

    def __repr__(self) -> str:
        return (
            f"CacheStats(hits={self.hits}, misses={self.misses}, "
            f"size={self.size}, maxsize={self.maxsize}, hit_rate={self.hit_rate:.3f})"
        )

    def __eq__(self, other: object) -> bool:
        if not isinstance(other, CacheStats):
            return NotImplemented
        return self.as_dict() == other.as_dict()

class LRUCache(Generic[K, V]):
    """Thread-safe LRU cache with optional per-entry TTL.

    Parameters
    ----------
    maxsize:
        Maximum number of entries the cache may hold.  When the cache is full
        and a new entry is inserted, the least-recently-used entry is evicted.
        ``None`` means unbounded.  Must be a positive integer or ``None``.
    ttl:
        Optional time-to-live in seconds.  When set, each entry expires ``ttl``
        seconds after it was stored.  Expired entries are reported as misses
        and are removed lazily on access (and eagerly by :meth:`evict_expired`).
        ``None`` (the default) means entries never expire.
    clock:
        Optional zero-argument callable returning the current time as a float.
        Defaults to :func:`time.monotonic`.  Injectable for testing.

    Raises
    ------
    ValueError
        If ``maxsize`` is not ``None`` and is not a positive integer.

        return self._clock()

    def _is_expired(self, expiry: Optional[float], now: float) -> bool:
        return expiry is not None and now >= expiry

    # -- public API -------------------------------------------------------

    def get(self, key: K, default: Optional[V] = None) -> Optional[V]:
        """Return the value for ``key`` or ``default`` if absent/expired.

        A cache hit moves the entry to the most-recently-used position and
        increments the hit counter; a miss increments the miss counter.
        """
        with self._lock:
            entry = self._cache.get(key)
            if entry is None:
                self._misses += 1
                return default

            now = self._now()

                # Lazily remove the expired entry.
                del self._cache[key]
                self._misses += 1
                return default
            # Mark as most-recently-used.
            self._cache.move_to_end(key)
            self._hits += 1
            return value

        """Store ``value`` under ``key``, evicting LRU entries if necessary."""
        with self._lock:

                self._cache.move_to_end(key)
                return

            self._cache.move_to_end(key)
            if self._maxsize is not None and len(self._cache) > self._maxsize:
                # Pop the least-recently-used item (first item).
