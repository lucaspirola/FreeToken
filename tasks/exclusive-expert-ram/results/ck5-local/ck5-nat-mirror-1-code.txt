                   "misses": self.misses,
                   "hit_rate": self.hits / (self.hits + self.misses) if (self.hits + self.misses) else 0.0,
                   "total_size": self._total_size,
                   "max_size": self.max_size,
                   "item_count": len(self._cache),
               }
